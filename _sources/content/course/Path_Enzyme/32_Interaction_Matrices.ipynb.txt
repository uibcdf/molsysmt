    "import molsysmt as msm\n",
    "from molsysmt import systems\n",
    "\n",
    "# Load our engineered complex (Enzyme + BHET substrate)\n",
    "petase = msm.convert('pdb:6EQE', to_form='molsysmt.MolSys', selection='molecule_type==\"protein\"')\n",
    "bhet = msm.convert('C1=CC(=CC=C1C(=O)OCCO)C(=O)OCCO', from_form='string:smiles', to_form='molsysmt.MolSys')\n",
    "molsys = msm.merge([petase, bhet])"
   ]

    "### 1. The Interaction Heatmap\n",
    "The function `get_contacts()` returns a boolean contact matrix: one entry per pair of atoms, `True` when the pair is closer than the threshold. Rendering that matrix with Matplotlib gives you a heatmap in which each pixel is one atom pair, so the bright bands are the regions of the enzyme that actually touch the substrate.\n",
    "\n",
    ":::{seealso} Related Tools & References\n",
    ":class: dropdown\n",
    "\n",
    "- {func}`molsysmt.structure.get_contacts`: the contact-matrix engine used throughout this module.\n",
    "- {func}`molsysmt.structure.get_distances`: the underlying distance calculation, when you need the continuous values rather than a threshold.\n",
    ":::"
   ]
  },
  {
   "cell_type": "code",
   "execution_count": null,
   "metadata": {},
   "outputs": [],
   "source": [
    "import matplotlib.pyplot as plt\n",
    "\n",
    "# Compute the contact matrix between the substrate and the enzyme\n",
    "contact_map = msm.structure.get_contacts(molsys, selection='molecule_type==\"small molecule\"',\n",
    "                                         selection_2='molecule_type==\"protein\"', threshold='0.6 nm')\n",
    "\n",
    "# Render the first structure of the contact map as a heatmap\n",
    "figure, axis = plt.subplots()\n",
    "axis.imshow(contact_map[0].astype(int), origin='lower', interpolation='nearest', aspect='auto')\n",
    "axis.set_xlabel('Enzyme atom index')\n",
    "axis.set_ylabel('Substrate atom index')\n",
    "plt.show()"
