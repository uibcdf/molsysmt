    "    simulation = msm.convert(molsys, to_form='openmm.Simulation', \n",
    "                             forcefield='AMBER14', water_model='TIP3P-FB',\n",
    "                             temperature='310 K')\n",
    "    print(f\"OpenMM Simulation ready: {type(simulation)}\")\n",
    "except Exception as e:\n",
    "    print(f\"OpenMM conversion failed: {e}\")"
   ]
