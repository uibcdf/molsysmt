    "import molsysmt as msm\n",
    "from molsysmt import systems\n",
    "\n",
    "# Load our solvated complex\n",
    "fibril = msm.convert('pdb:2BEG', to_form='molsysmt.MolSys')\n",
    "peptide = msm.build.build_peptide('KLVFF')\n",
    "molsys = msm.merge([fibril, peptide])\n",
    "\n",
    "# Add water to see the hydration shell\n",
    "solvated_molsys = msm.build.solvate(molsys, clearance='1.0 nm')"
   ]
  },
  {
   "cell_type": "markdown",
   "metadata": {},
   "source": [
    "### 1. Finding Neighboring Residues\n",
    "The function `get_neighbors()` returns a list of elements that are within a certain distance from your target. Let's find which protein residues are within 0.5 nm of our peptide."
   ]
  },
  {
   "cell_type": "code",
   "execution_count": null,
   "metadata": {},
   "outputs": [],
   "source": [
    "# Find residues of the protein (selection_2) near the peptide (selection)\n",
    "neighbors = msm.get_neighbors(solvated_molsys, selection='molecule_type==\"peptide\"', \n",
    "                              selection_2='molecule_type==\"protein\"', \n",
    "                              threshold='0.5 nm', element='group')\n",
    "\n",
    "print(f\"Found {len(neighbors[0])} neighboring residues in the fibril.\")\n",
    "res_names = msm.get(solvated_molsys, element='group', selection=neighbors[0], group_name=True)\n",
    "print(f\"Neighborhood residues: {set(res_names)}\")"
   ]
  },
